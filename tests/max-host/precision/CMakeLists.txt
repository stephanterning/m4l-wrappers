cmake_minimum_required(VERSION 3.25)
project(MaxHostPrecision VERSION 1.0.0 LANGUAGES C CXX)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "")
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "")
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "")
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "")
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "")
file(TO_CMAKE_PATH "${VST3_SDK}" VST3_SDK)
add_subdirectory("${VST3_SDK}" sdk EXCLUDE_FROM_ALL)
function(add_probe target source identifier)
    smtg_add_vst3plugin(${target} ${source}
        "${VST3_SDK}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
    target_link_libraries(${target} PRIVATE sdk)
    if(APPLE)
        target_sources(${target} PRIVATE "${VST3_SDK}/public.sdk/source/main/macmain.cpp")
        smtg_target_set_bundle(${target} BUNDLE_IDENTIFIER "org.m4lwrappers.test.${identifier}" COMPANY_NAME "M4L Wrappers")
    elseif(WIN32)
        target_sources(${target} PRIVATE "${VST3_SDK}/public.sdk/source/main/dllmain.cpp")
    endif()
endfunction()
add_probe(MaxHostPrecision precision.cpp precision)
add_probe(MaxHostSetup setup.cpp setup)
