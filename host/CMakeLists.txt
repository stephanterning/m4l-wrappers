cmake_minimum_required(VERSION 3.25)
project(m4l_max_host_output LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_VS_PLATFORM_NAME STREQUAL "x64")
    message(FATAL_ERROR "Configure for Windows x64 with Visual Studio")
endif()
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(MAX_SDK_BASE "${root}/deps/max-sdk-base" CACHE PATH "Cycling '74 Max SDK base checkout")
set(max "${MAX_SDK_BASE}/c74support/max-includes")
set(msp "${MAX_SDK_BASE}/c74support/msp-includes")
add_library(output MODULE output.cpp)
target_compile_features(output PRIVATE cxx_std_17)
target_compile_definitions(output PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(output PRIVATE /utf-8)
set_property(TARGET output PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(output PRIVATE "${root}/shared" "${max}" "${msp}")
target_link_libraries(output PRIVATE "${max}/x64/MaxAPI.lib" "${msp}/x64/MaxAudio.lib")
set_target_properties(output PROPERTIES PREFIX "" SUFFIX ".mxe64" OUTPUT_NAME "m4lwrappergainstage~"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/externals"
    LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/externals")
